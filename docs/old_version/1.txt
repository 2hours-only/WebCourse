### 模块间数据流向
```
用户操作
    ↓
ui.panel.getUserInput() → UserPreference
    ↓
main.js 接收 → recommend.recommend(userPreference, cinema)
    ↓
返回推荐座位 → canvas.renderer.updateSeat(seat)
    ↓
用户选择 → canvas.interaction.onSeatClick → 更新 Seat.status
    ↓
ui.panel.setSelectedSeats(seats) → 显示已选
    ↓
用户确认 → order.confirm() → storage.saveOrder(order)
    ↓
ui.dialog.showSuccess("购票成功")
```

# 一、功能与界面效果目标

整个系统的流程目标如下：

```
用户输入需求
    ↓
系统智能推荐
    ↓
用户选择座位
    ↓
生成订单
    ↓
完成购票
```

不要设计很多页面，围绕上面的"快速选座"流程设计即可，作业要求**Don't Make Me Think.**

页面采用如下布局：

```
┌──────────────────────────────────────────────────────┐
│ Header                                               │
├──────────────┬──────────────────────┬────────────────┤
│              │                      │                │
│  左侧控制区    │      座位图          │   右侧信息区域   │
│              │                      │                │
├──────────────┴──────────────────────┴────────────────┤
│ Footer                                               │
└──────────────────────────────────────────────────────┘
```

具体的：

左侧：

- 用户信息输入
- 推荐按钮
- 无障碍设置

中间：

- Canvas影院
- 座位交互
- 热力图显示

右侧：

- 推荐信息
- 当前选座
- 订单信息
- 购票操作

# 二、整体架构与对象模型

系统架构分层，各层职责单一, 方便大家各自独立推进。

```
UI层
↓
交互层
↓
业务逻辑层
↓
推荐算法层
↓
Canvas渲染层
↓
数据层
↓
LocalStorage存储
```

### 整个系统由 main.js 负责调用，其余层彼此之间-不允许-相互调用!!! (>ᴗ<)

正确流程：

```
main
    │
    ├── recommend
    ├── canvas
    ├── storage
    └── ui
    ...
```

错误流程：

```
recommend
↓
canvas
↓
canvas
↓
storage
↓
storage
↓
ui
```

---

整个项目包含四种核心对象。

## Cinema

表示整个影院。

包含：

- 行数
- 列数
- 屏幕位置
- 所有座位

Canvas绘制影院时读取Cinema。
推荐算法也读取Cinema。

## Seat

表示一个座位。

包含：

- 行号
- 列号
- 当前状态（空闲/已售/已选）
- 是否推荐
- 热度
- 评分
- 价格

**Seat全局唯一**, Canvas、推荐、订单全部操作同一个Seat对象，**不允许复制数据**。

## UserPreference

表示用户输入。

包含：

- 年龄
- 人数
- 观影类型
- 是否开启无障碍模式
- （可扩展预算等）

推荐算法只读取UserPreference。

## Order

表示订单。

包含：

- 订单编号
- 座位列表
- 金额
- 时间
- 状态

LocalStorage最终保存Order。

## app

整个项目入口。

负责：

- 初始化项目
- 初始化数据
- 注册事件
- 协调各模块工作

不负责任何具体业务逻辑。

#### app/main.js

整个系统的控制中心，只做流程调度。

**负责**

- 初始化 Cinema（从 data 读取）

- 初始化 UI

- 绑定事件监听

- 调用 recommend 获取推荐结果

- 调用 canvas 更新画面

- 调用 storage 保存订单

- 协调所有模块数据流

**禁止**

- 写推荐算法

- 写Canvas绘制逻辑

- 写UI组件细节

- 直接操作LocalStorage

import { Cinema } from "../core/cinema.js";
import { Order } from "../core/order.js";
import { CanvasRenderer } from "../canvas/renderer.js";

import { HeatmapRenderer } from "../canvas/heatmap.js";
import { RecommendEngine } from "../recommend/recommend.js";
import { UIPanel } from "../ui/panel.js";
import { StorageManager } from "../storage/storage.js";
import { EventBus } from "../utils/eventBus.js";

class MainController {
  constructor() {

    this.eventBus = new EventBus();
    this.selectedSeats = [];
    this.init();
  }

  init() {
    console.log("[Main] App Starting...");

    this.cinema = new Cinema(8, 10, "top");
    this.storage = new StorageManager();

    this.uiPanel = new UIPanel(
      document.querySelector(".main-container"),
      this.eventBus,
    );

    const canvasEl = this.getElementByIdOrCreate("cinema-canvas");

    this.resizeCanvas(canvasEl);
    window.addEventListener("resize", () => this.handleResize(canvasEl));

    this.renderer = new CanvasRenderer(canvasEl, this.cinema);

    this.heatmapRenderer = new HeatmapRenderer(this.renderer);

    this.interaction = new InteractionHandler(
      canvasEl,
      this.renderer,
      this.eventBus,
    );

    this.recommendEngine = new RecommendEngine();

    this.registerEvents();

    this.renderer.render();
  }

  getElementByIdOrCreate(id) {
    let el = document.getElementById(id);
    if (!el) {
      el = document.createElement("canvas");
      el.id = id;
      document.querySelector(".cinema-container").appendChild(el);
    }
    return el;
  }

  handleResize(canvasEl) {
    this.resizeCanvas(canvasEl);

    this.renderer.render();
    this.heatmapRenderer.render(); 
  }

  resizeCanvas(canvasEl) {

    const container = document.querySelector(".cinema-container");
    if (container) {
      canvasEl.width = container.clientWidth;

      canvasEl.height = Math.min(container.clientWidth * 0.6, 600);
    }
  }

  registerEvents() {

    this.eventBus.on("user:recommend", (userPref) => {
      console.log("[Main] Received user:recommend event", userPref);
      const seats = this.recommendEngine.recommend(userPref, this.cinema);

      seats.forEach((s) => s.setRecommended(true));

      this.renderer.renderSeats(seats);
      this.heatmapRenderer.generate(seats); 
      this.heatmapRenderer.render(); 
      this.uiPanel.setRecommendation(seats);
    });

    this.eventBus.on("seat:clicked", (seat) => {
      console.log("[Main] Received seat:clicked event", seat);

      if (seat.status === "available") {
        seat.setStatus("selected");
        this.selectedSeats.push(seat);

        this.renderer.updateSeat(seat);
        this.uiPanel.setSelectedSeats(this.selectedSeats);
      } else if (seat.status === "selected") {

        seat.setStatus("available");
        this.selectedSeats = this.selectedSeats.filter((s) => s !== seat);
        this.renderer.updateSeat(seat);
        this.uiPanel.setSelectedSeats(this.selectedSeats);
      }
    });

    this.eventBus.on("user:purchase", () => {
      console.log("[Main] Received user:purchase event");
      if (this.selectedSeats.length === 0) return;

      const order = new Order(this.selectedSeats);
      order.calculateAmount();

      this.storage.saveOrder(order);

      this.uiPanel.setOrderInfo(order);

      this.selectedSeats.forEach((s) => s.setStatus("sold"));
      this.renderer.renderSeats(this.selectedSeats);

      this.selectedSeats = [];
      this.uiPanel.setSelectedSeats([]);
    });

    this.eventBus.on("accessibility:change", (config) => {
      console.log("[Main] Accessibility settings changed", config);

      this.renderer.render();
    });
  }
}

new MainController();

import { Seat } from "./seat.js";

export class Cinema {
  constructor(rows, cols, screenPosition) {
    console.log("[Core] Cinema Initializing...");
    this.rows = rows;
    this.cols = cols;
    this.screenPosition = screenPosition;
    this.seats = [];
    this._initSeats();
  }

  _initSeats() {
    for (let r = 0; r < this.rows; r++) {
      this.seats[r] = [];
      for (let c = 0; c < this.cols; c++) {
        this.seats[r][c] = new Seat(r, c);
      }
    }
  }

  getSeat(row, col) {
    console.log(`[Core] Cinema getSeat(${row}, ${col})`);
    if (row >= 0 && row < this.rows && col >= 0 && col < this.cols) {
      return this.seats[row][col];
    }
    return null;
  }

  getRow(row) {
    console.log(`[Core] Cinema getRow(${row})`);
    if (row >= 0 && row < this.rows) {
      return this.seats[row];
    }
    return [];
  }

  getAllSeats() {
    console.log("[Core] Cinema getAllSeats");
    return this.seats.flat();
  }

  getAvailableSeats() {
    console.log("[Core] Cinema getAvailableSeats");
    return this.getAllSeats().filter((s) => s.status === "available");
  }
}

---=== js/ui/panel.js ===---
export class UIPanel {
  constructor(container, eventBus) {
    this.container = container;
    this.eventBus = eventBus; 
    console.log("[UI] Panel created");

    this.bindDOMEvents();
  }

  bindDOMEvents() {
    const recommendBtn = this.container.querySelector("#recommend-btn");
    if (recommendBtn) {
      recommendBtn.addEventListener("click", () => {
        const userInput = this.getUserInput();

        this.eventBus.emit("user:recommend", userInput);
      });
    }

    const purchaseBtn = this.container.querySelector("#purchase-btn");
    if (purchaseBtn) {
      purchaseBtn.addEventListener("click", () => {
        this.eventBus.emit("user:purchase");
      });
    }
  }

  getUserInput() {
    console.log("[UI] getUserInput");
    const ageSelect = document.getElementById("age-select");
    const countInput = document.getElementById("people-count");

    return {
      age: ageSelect ? ageSelect.value : "adult",
      count: countInput ? parseInt(countInput.value) : 2,
      type: "action",
    };
  }

  setRecommendation(seats) {
    const resultDiv = document.getElementById("recommend-result");
    resultDiv.innerHTML = `<p>推荐座位: ${seats.map((s) => `${s.row}排${s.col}座`).join(", ")}</p>`;
  }

  setSelectedSeats(seats) {
    const selectedDiv = document.getElementById("selected-seats");
    selectedDiv.innerHTML = `<p>已选: ${seats.map((s) => `${s.row}排${s.col}座`).join(", ")}</p>`;

    const purchaseBtn = document.getElementById("purchase-btn");
    if (purchaseBtn) purchaseBtn.disabled = seats.length === 0;
  }

  setOrderInfo(order) {
    const orderDiv = document.getElementById("order-info");
    orderDiv.innerHTML = `<p>订单成功! 总额: ¥${order.amount}</p>`;
  }
}

---=== js/utils/eventBus.js ===---
export class EventBus {
  constructor() {
    this.events = {};
  }
  on(eventName, callback) {
    if (!this.events[eventName]) this.events[eventName] = [];
    this.events[eventName].push(callback);
  }
  emit(eventName, payload) {
    if (this.events[eventName]) {
      this.events[eventName].forEach((cb) => cb(payload));
    }
  }
}
